2-Load and display the original image

path=r"C:\Users\hp\Downloads\devanagari+handwritten+character+dataset (1)\DevanagariHandwrittenCharacterDataset\Train"

classes = os.listdir(path)
classes = [c for c in classes if os.path.isdir(os.path.join(path,c))]

folder = os.path.join(path,classes[0])
file = os.listdir(folder)[0]
img_path = os.path.join(folder,file)

img = cv2.imread(img_path)

plt.imshow( cv2.cvtColor(img, cv2.COLOR_BGR2RGB))
plt.title("original image")
plt.axis("off")
plt.show()


3-Check image properties

print("Shape:", img.shape)
print("Height:", img.shape[0])
print("Width:", img.shape[1])
print("Image channels: " , img.shape[2])

4-Convert to grayscale

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

plt.imshow(gray, cmap="gray")
plt.title("Grayscale Image")
plt.axis("off")
plt.show()

5-Apply binary thresholding

_, binary = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)

plt.imshow(binary, cmap="gray")
plt.title("Binary Image")
plt.axis("off")
plt.show()

6-Apply adaptive thresholding

adaptive = cv2.adaptiveThreshold(
    gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
    cv2.THRESH_BINARY, 11, 2
)

plt.imshow(adaptive, cmap="gray")
plt.title("Adaptive Threshold")
plt.axis("off")
plt.show()

7-Detect edges

edges = cv2.Canny(gray, 100, 200)

plt.imshow(edges, cmap="gray")
plt.title("Edge Detection")
plt.axis("off")
plt.show()

8-Save the processed image

cv2.imwrite("processed_img.png", binary)
print("Processed image saved")

9-Resize the image

resized = cv2.resize(gray, (32, 32))

plt.imshow(resized, cmap="gray")
plt.title("Resized Image")
plt.axis("off")
plt.show()

10-Remove noise

denoised = cv2.medianBlur(resized, 3)

plt.imshow(denoised, cmap="gray")
plt.title("Noise Removed")
plt.axis("off")
plt.show()


11-Compare all processed images

images = [img, gray, resized, denoised, binary, adaptive, edges]
titles = ["Original", "Gray", "Resized", "Denoised",
          "Binary", "Adaptive", "Edges"]

plt.figure(figsize=(12, 7))

for i in range(7):
    plt.subplot(3, 3, i + 1)

    if i == 0:
        plt.imshow(cv2.cvtColor(images[i], cv2.COLOR_BGR2RGB))
    else:
        plt.imshow(images[i], cmap="gray")

    plt.title(titles[i])
    plt.axis("off")

plt.tight_layout()
plt.show()
